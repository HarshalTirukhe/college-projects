# 1. Check Git version
git --version

# 2. Initialize a Git repository
git init

# 3. Check the remote repository
git remote -v

# 4. Add the GitHub remote repository
git remote add origin https://github.com/harshal-081/vsec.git

# If "origin already exists" appears, use this instead:
git remote set-url origin https://github.com/harshal-081/vsec.git

# 5. Fetch files/branches from GitHub
git fetch origin

# 6. See available branches
git branch
git branch -a

# 7. Rename the current branch to main
git branch -M main

# 8. Get the remote main branch
git checkout -b main origin/main

# 9. Stage all files
git add .

# 10. Check the current Git status
git status

# 11. Commit staged files
git commit -m "Assignment1"

# 12. Pull changes from GitHub
git pull origin main --no-rebase --allow-unrelated-histories

# 13. If there is a merge conflict:
# First fix the conflicting files manually, then:
git add .
git commit -m "Resolved merge conflicts"

# 14. Push your local main branch to GitHub
git push origin main

# 15. If Git says remote contains work you don't have:
git pull origin main --no-rebase
git push origin main

# 16. See commit history
git log --oneline

# 17. See which files are staged/unstaged
git status

# 18. Unstage a file
git restore --staged filename

# 19. Unstage everything
git restore --staged .

# 20. Undo the latest commit but keep the files
git reset HEAD~1

# 21. Check configured remote
git remote -v
